package za.co.sprinta.app.ads

import android.app.Activity
import android.content.Context
import com.google.android.gms.ads.AdRequest
import com.google.android.gms.ads.LoadAdError
import com.google.android.gms.ads.MobileAds
import com.google.android.gms.ads.interstitial.InterstitialAd
import com.google.android.gms.ads.interstitial.InterstitialAdLoadCallback
import com.google.android.ump.UserMessagingPlatform

object AdManager {
    const val BANNER_AD_UNIT_ID = "ca-app-pub-5833611172742289/3073358578"
    const val INTERSTITIAL_AD_UNIT_ID = "ca-app-pub-5833611172742289/4286365295"

    private var interstitialAd: InterstitialAd? = null
    private var isInitializing = false

    fun gatherConsentAndInitialize(activity: Activity) {
        if (isInitializing) return
        isInitializing = true

        val consentInformation = UserMessagingPlatform.getConsentInformation(activity)
        val params = com.google.android.ump.ConsentRequestParameters.Builder().build()

        consentInformation.requestConsentInfoUpdate(
            activity,
            params,
            {
                UserMessagingPlatform.loadAndShowConsentFormIfRequired(activity) { _ ->
                    initMobileAds(activity.applicationContext)
                }
            },
            { _ ->
                initMobileAds(activity.applicationContext)
            }
        )
    }

    private fun initMobileAds(context: Context) {
        MobileAds.initialize(context) {
            preloadInterstitial(context)
        }
    }

    fun preloadInterstitial(context: Context) {
        val adRequest = AdRequest.Builder().build()
        InterstitialAd.load(
            context,
            INTERSTITIAL_AD_UNIT_ID,
            adRequest,
            object : InterstitialAdLoadCallback() {
                override fun onAdLoaded(ad: InterstitialAd) {
                    interstitialAd = ad
                }

                override fun onAdFailedToLoad(error: LoadAdError) {
                    interstitialAd = null
                }
            }
        )
    }

    fun showInterstitialIfReady(activity: Activity, onAdDismissed: () -> Unit) {
        val ad = interstitialAd
        if (ad != null) {
            ad.fullScreenContentCallback = object : com.google.android.gms.ads.FullScreenContentCallback() {
                override fun onAdDismissedFullScreenContent() {
                    interstitialAd = null
                    preloadInterstitial(activity.applicationContext)
                    onAdDismissed()
                }

                override fun onAdFailedToShowFullScreenContent(p0: com.google.android.gms.ads.AdError) {
                    interstitialAd = null
                    onAdDismissed()
                }
            }
            ad.show(activity)
        } else {
            onAdDismissed()
        }
    }
}
package za.co.sprinta.app.billing

import android.app.Activity
import android.app.Application
import androidx.lifecycle.AndroidViewModel
import com.android.billingclient.api.*
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow

class SprintaBillingViewModel(application: Application) : AndroidViewModel(application), PurchasesUpdatedListener {

    private val billingClient = BillingClient.newBuilder(application)
        .setListener(this)
        .enablePendingPurchases()
        .build()

    private val _isProUser = MutableStateFlow(false)
    val isProUser: StateFlow<Boolean> = _isProUser

    private val _proProductDetails = MutableStateFlow<ProductDetails?>(null)
    val proProductDetails: StateFlow<ProductDetails?> = _proProductDetails

    init {
        connectToPlayBilling()
    }

    private fun connectToPlayBilling() {
        billingClient.startConnection(object : BillingClientStateListener {
            override fun onBillingSetupFinished(billingResult: BillingResult) {
                if (billingResult.responseCode == BillingClient.BillingResponseCode.OK) {
                    querySubscriptionProducts()
                    checkExistingPurchases()
                }
            }
            override fun onBillingServiceDisconnected() {
                // Retry connection logic here in production
            }
        })
    }

    private fun querySubscriptionProducts() {
        val queryProductDetailsParams = QueryProductDetailsParams.newBuilder()
            .setProductList(
                listOf(
                    QueryProductDetailsParams.Product.newBuilder()
                        .setProductId("sprinta_pro_monthly") // This must match your Google Play Console ID
                        .setProductType(BillingClient.ProductType.SUBS)
                        .build()
                )
            )
            .build()

        billingClient.queryProductDetailsAsync(queryProductDetailsParams) { _, productDetailsList ->
            if (productDetailsList.isNotEmpty()) {
                _proProductDetails.value = productDetailsList[0]
            }
        }
    }

    fun launchBillingFlow(activity: Activity) {
        val productDetails = _proProductDetails.value ?: return
        val offerToken = productDetails.subscriptionOfferDetails?.get(0)?.offerToken ?: return

        val productDetailsParamsList = listOf(
            BillingFlowParams.ProductDetailsParams.newBuilder()
                .setProductDetails(productDetails)
                .setOfferToken(offerToken)
                .build()
        )

        val billingFlowParams = BillingFlowParams.newBuilder()
            .setProductDetailsParamsList(productDetailsParamsList)
            .build()

        billingClient.launchBillingFlow(activity, billingFlowParams)
    }

    override fun onPurchasesUpdated(billingResult: BillingResult, purchases: MutableList<Purchase>?) {
        if (billingResult.responseCode == BillingClient.BillingResponseCode.OK && purchases != null) {
            for (purchase in purchases) {
                acknowledgePurchase(purchase)
            }
        }
    }

    private fun acknowledgePurchase(purchase: Purchase) {
        if (purchase.purchaseState == Purchase.PurchaseState.PURCHASED) {
            if (!purchase.isAcknowledged) {
                val acknowledgePurchaseParams = AcknowledgePurchaseParams.newBuilder()
                    .setPurchaseToken(purchase.purchaseToken)
                    .build()
                billingClient.acknowledgePurchase(acknowledgePurchaseParams) { billingResult ->
                    if (billingResult.responseCode == BillingClient.BillingResponseCode.OK) {
                        _isProUser.value = true
                    }
                }
            } else {
                _isProUser.value = true
            }
        }
    }

    private fun checkExistingPurchases() {
        billingClient.queryPurchasesAsync(
            QueryPurchasesParams.newBuilder().setProductType(BillingClient.ProductType.SUBS).build()
        ) { _, purchases ->
            val hasActiveSub = purchases.any { it.purchaseState == Purchase.PurchaseState.PURCHASED }
            _isProUser.value = hasActiveSub
        }
    }
}

    val recordType: String,

    val workoutId: String,

@Dao
@JvmSuppressWildcards
interface WorkoutDao {
    @Query("SELECT * FROM workouts WHERE userId = :userId ORDER BY startTimeIso DESC")
    fun observeWorkouts(userId: String): Flow<List<WorkoutEntity>>

    @Query("SELECT * FROM workouts WHERE isSynced = 0 ORDER BY startTimeIso ASC")
    suspend fun getUnsynced(): List<WorkoutEntity>

    @Query("UPDATE workouts SET isSynced = 1 WHERE workoutId = :workoutId")
    suspend fun markSynced(workoutId: String): Int

    @Query("SELECT * FROM workouts WHERE userId = :userId ORDER BY startTimeIso DESC")
    suspend fun getWorkouts(userId: String): List<WorkoutEntity>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertWorkout(workout: WorkoutEntity): Long
}

@Dao
@JvmSuppressWildcards
interface PersonalRecordDao {
    @Query("SELECT * FROM personal_records WHERE userId = :userId")
    suspend fun getRecords(userId: String): List<PersonalRecordEntity>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertRecord(record: PersonalRecordEntity): Long
}

@Database(
    entities = [WorkoutEntity::class, PersonalRecordEntity::class],
    version = 2,
    exportSchema = false
)
abstract class SprintaDatabase : RoomDatabase() {
    abstract fun workoutDao(): WorkoutDao
    abstract fun personalRecordDao(): PersonalRecordDao
}

class SprintaRepository(
    private val database: SprintaDatabase,
    private val api: SprintaApiService,
    private val session: SecureSessionStore
) {
    private val gson = Gson()

    fun observeWorkouts(userId: String): Flow<List<WorkoutEntity>> =
        database.workoutDao().observeWorkouts(userId)

    suspend fun register(name: String, email: String, password: String): Result<AuthResponse> =
        executeAuth { api.register(RegisterRequest(name.trim(), email.trim(), password)) }

    suspend fun login(email: String, password: String): Result<AuthResponse> =
        executeAuth { api.login(LoginRequest(email.trim(), password)) }

    private suspend fun executeAuth(call: suspend () -> Response<AuthResponse>): Result<AuthResponse> {
        return try {
            val response = call()
            if (response.isSuccessful && response.body()?.user != null) {
                Result.success(response.body()!!)
            } else {
                Result.failure(IllegalStateException("Authentication request failed: ${response.code()}"))
            }
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    fun saveSession(response: AuthResponse) {
        val user = requireNotNull(response.user)
        session.save(user.id, user.name, user.email, response.token)
    }

    fun clearSession() = session.clear()

    fun currentSession() = session.load()

    suspend fun saveWorkout(workout: WorkoutEntity): List<String> {
        database.workoutDao().insertWorkout(workout)
        val existing = database.personalRecordDao().getRecords(workout.userId)
        val newRecords = PersonalRecordUtils.detectNewRecords(workout, existing)
        newRecords.forEach { record ->
            database.personalRecordDao().insertRecord(
                PersonalRecordUtils.toEntity(record, workout.userId, workout.workoutId)
            )
        }
        return newRecords.map { it.type }
    }

    suspend fun getUnsyncedWorkouts(): List<WorkoutEntity> = database.workoutDao().getUnsynced()

    suspend fun markSynced(workoutId: String) = database.workoutDao().markSynced(workoutId)

    suspend fun refreshFromServer(): Result<Int> {
        val current = session.load() ?: return Result.failure(IllegalStateException("No session"))
        return try {
            val response = api.getUserWorkouts(
                "Bearer ${current["token"]}",
                current["id"]!!
            )
            if (!response.isSuccessful) {
                Result.failure(IllegalStateException("History request failed: ${response.code()}"))
            } else {
                val items = response.body().orEmpty()
                items.forEach { dto ->
                    val entity = WorkoutEntity(
                        workoutId = dto.workoutId,
                        userId = dto.userId,
                        distanceMeters = dto.distanceMeters,
                        durationSeconds = dto.durationSeconds,
                        averagePaceSecondsPerKm = dto.averagePaceSecondsPerKm,
                        startTimeIso = dto.startTimeIso,
                        totalElevation = dto.totalElevation,
                        notes = dto.notes,
                        routePointsJson = gson.toJson(dto.routePoints),
                        isSynced = true
                    )
                    database.workoutDao().insertWorkout(entity)
                }
                Result.success(items.size)
            }
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    fun entityToDto(entity: WorkoutEntity): WorkoutDto {
        val points: List<RoutePoint> = runCatching {
            gson.fromJson(entity.routePointsJson, Array<RoutePoint>::class.java)?.toList() ?: emptyList()
        }.getOrDefault(emptyList())

        return WorkoutDto(
            workoutId = entity.workoutId,
            userId = entity.userId,
            distanceMeters = entity.distanceMeters,
            durationSeconds = entity.durationSeconds,
            averagePaceSecondsPerKm = entity.averagePaceSecondsPerKm,
            startTimeIso = entity.startTimeIso,
            totalElevation = entity.totalElevation,
            notes = entity.notes,
            routePoints = points.map {
                za.co.sprinta.app.model.LocationPointDto(
                    it.latitude,
                    it.longitude,
                    it.altitude,
                    it.timestamp
                )
            }
        )
    }
}

import android.content.Context
import androidx.work.Constraints
import androidx.work.CoroutineWorker
import androidx.work.NetworkType
import androidx.work.OneTimeWorkRequestBuilder
import androidx.work.WorkManager
import androidx.work.WorkerParameters
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import za.co.sprinta.app.AppGraph

class SyncWorker(
    appContext: Context,
    workerParams: WorkerParameters
) : CoroutineWorker(appContext, workerParams) {

    override suspend fun doWork(): Result = withContext(Dispatchers.IO) {
        val unsynced = AppGraph.repository.getUnsyncedWorkouts()
        if (unsynced.isEmpty()) return@withContext Result.success()

        val token = AppGraph.session.load()?.get("token")

        return@withContext try {
            for (workout in unsynced) {
                if (token != null) {
                    val response = AppGraph.api.syncWorkout(
                        "Bearer $token",
                        AppGraph.repository.entityToDto(workout)
                    )
                    if (response.isSuccessful) {
                        AppGraph.repository.markSynced(workout.workoutId)
                    } else if (response.code() in 500..599) {
                        return@withContext Result.retry()
                    }
                } else {
                    AppGraph.repository.markSynced(workout.workoutId)
                }
            }
            Result.success()
        } catch (_: Exception) {
            for (workout in unsynced) {
                AppGraph.repository.markSynced(workout.workoutId)
            }
            Result.success()
        }
    }

    companion object {
        fun schedule(context: Context) {
            val constraints = Constraints.Builder()
                .setRequiredNetworkType(NetworkType.CONNECTED)
                .build()

            val request = OneTimeWorkRequestBuilder<SyncWorker>()
                .setConstraints(constraints)
                .build()

            WorkManager.getInstance(context).enqueue(request)
        }
    }
}

private val Context.dataStore by preferencesDataStore(name = "sprinta_settings")

data class AppPreferences(
    val darkMode: Boolean = false,
    val useMiles: Boolean = false,
    val audioIntervalKm: Int = 1,
    val language: String = "en"
)

class PreferencesStore(private val context: Context) {
    private val darkKey = booleanPreferencesKey("dark_mode")
    private val milesKey = booleanPreferencesKey("use_miles")
    private val audioKey = intPreferencesKey("audio_interval")
    private val languageKey = stringPreferencesKey("language")

    val preferences: Flow<AppPreferences> = context.dataStore.data.map { p ->
        AppPreferences(
            darkMode = p[darkKey] ?: false,
            useMiles = p[milesKey] ?: false,
            audioIntervalKm = p[audioKey] ?: 1,
            language = p[languageKey] ?: "en"
        )
    }

    suspend fun setDarkMode(value: Boolean) = context.dataStore.edit { it[darkKey] = value }
    suspend fun setUseMiles(value: Boolean) = context.dataStore.edit { it[milesKey] = value }
    suspend fun setAudioInterval(value: Int) = context.dataStore.edit { it[audioKey] = value }
    suspend fun setLanguage(value: String) = context.dataStore.edit { it[languageKey] = value }
}

class SecureSessionStore(context: Context) {
    private val prefs = run {
        val masterKey = MasterKey.Builder(context)
            .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
            .build()
        EncryptedSharedPreferences.create(
            context,
            "sprinta_secure_session",
            masterKey,
            EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
            EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
        )
    }

    fun save(id: String, name: String, email: String, token: String) {
        prefs.edit()
            .putString("id", id)
            .putString("name", name)
            .putString("email", email)
            .putString("token", token)
            .apply()
    }

    fun load(): Map<String, String>? {
        val id = prefs.getString("id", null) ?: return null
        val name = prefs.getString("name", null) ?: return null
        val email = prefs.getString("email", null) ?: return null
        val token = prefs.getString("token", null) ?: return null
        return mapOf("id" to id, "name" to name, "email" to email, "token" to token)
    }

    fun clear() = prefs.edit().clear().apply()
}

class LocationTrackingService : Service(), TextToSpeech.OnInitListener {
    private lateinit var fusedLocationClient: FusedLocationProviderClient
    private var tts: TextToSpeech? = null

    private val serviceScope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    private var audioIntervalKm: Int = 1
    private var lastAnnouncedKm: Int = 0

    private val callback = object : LocationCallback() {
        override fun onLocationResult(result: LocationResult) {
            result.locations.forEach { location: Location ->
                // GPS Jitter Filter: Ignore accuracy worse than 15 meters
                if (location.accuracy > 15f) return@forEach

                val currentPoints = TrackingStore.state.value.points
                val previous = currentPoints.lastOrNull()

                if (previous != null) {
                    val timeDiffSeconds = (location.time - previous.timestamp) / 1000.0
                    val dist = GeoUtils.haversineMeters(
                        previous.latitude, previous.longitude,
                        location.latitude, location.longitude
                    )
                    // Speed Filter: Ignore impossible human speeds (> 15 meters/sec / 54 km/h)
                    if (timeDiffSeconds > 0 && (dist / timeDiffSeconds) > 15.0) return@forEach
                }

                TrackingStore.addLocation(location)
                checkAndAnnounceSplit()
            }
        }
    }

    override fun onCreate() {
        super.onCreate()
        fusedLocationClient = LocationServices.getFusedLocationProviderClient(this)
        tts = TextToSpeech(this, this)

        // Observe user settings for Audio Updates
        serviceScope.launch {
            AppGraph.preferences.preferences.collect { prefs ->
                audioIntervalKm = prefs.audioIntervalKm
            }
        }

        createNotificationChannel()
        startTrackingForeground()
    }

    private fun checkAndAnnounceSplit() {
        if (audioIntervalKm == 0) return // Audio updates turned off in settings

        val currentKm = (TrackingStore.state.value.distanceMeters / 1000).toInt()
        if (currentKm >= lastAnnouncedKm + audioIntervalKm) {
            lastAnnouncedKm = currentKm
            val pace = GeoUtils.formatPace(TrackingStore.state.value.paceSecondsPerKm)

            val message = "$currentKm kilometer${if (currentKm > 1) "s" else ""} completed. Average pace: $pace."
            tts?.speak(message, TextToSpeech.QUEUE_FLUSH, null, null)
        }
    }

    override fun onInit(status: Int) {
        if (status == TextToSpeech.SUCCESS) {
            tts?.language = Locale.getDefault()
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START -> startUpdates()
            ACTION_STOP -> stopTracking()
        }
        return START_STICKY
    }

    private fun startTrackingForeground() {
        val notification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("SPRINTA")
            .setContentText("Tracking your run in real-time...")
            .setSmallIcon(android.R.drawable.ic_media_play)
            .setOngoing(true)
            .build()

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            startForeground(NOTIFICATION_ID, notification, android.content.pm.ServiceInfo.FOREGROUND_SERVICE_TYPE_LOCATION)
        } else {
            startForeground(NOTIFICATION_ID, notification)
        }
    }

    private fun startUpdates() {
        TrackingStore.start()
        lastAnnouncedKm = 0

        // Battery Saver Location Request
        val request = LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, 2000L)
            .setMinUpdateIntervalMillis(1000L)
            .setMinUpdateDistanceMeters(2.0f) // Only fetch if user moved 2 meters
            .build()

        try {
            fusedLocationClient.requestLocationUpdates(request, callback, Looper.getMainLooper())
        } catch (e: SecurityException) {
            stopSelf()
        }
    }

    private fun stopTracking() {
        fusedLocationClient.removeLocationUpdates(callback)
        TrackingStore.stop()
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
    }

    override fun onDestroy() {
        serviceScope.cancel()
        tts?.stop()
        tts?.shutdown()
        fusedLocationClient.removeLocationUpdates(callback)
        super.onDestroy()
    }

data class TrackingState(
    val isTracking: Boolean = false,
    val distanceMeters: Double = 0.0,
    val durationSeconds: Long = 0L,
    val currentPaceSecondsPerKm: Double = 0.0,
    val routeCoordinates: List<Pair<Double, Double>> = emptyList()
)

class TrackingService : Service() {

    private lateinit var fusedLocationClient: FusedLocationProviderClient
    private var lastLocation: Location? = null

    private val callback = object : LocationCallback() {
        override fun onLocationResult(result: LocationResult) {
            result.locations.forEach { location ->
                if (LocationUtils.isReliableLocation(location, lastLocation)) {
                    lastLocation = location
                    TrackingStore.addLocation(location)
                    updateTrackingState()
                }
            }
        }
    }

    override fun onCreate() {
        super.onCreate()
        fusedLocationClient = LocationServices.getFusedLocationProviderClient(this)
        createNotificationChannel()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START -> startTracking()
            ACTION_STOP -> stopTracking()
        }
        return START_STICKY
    }

    private fun startTracking() {
        startForegroundServiceNotification()
        TrackingStore.start()
        updateTrackingState()

        val locationRequest = LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, 2000L)
            .setMinUpdateIntervalMillis(1000L)
            .setMinUpdateDistanceMeters(2.0f)
            .build()

        try {
            fusedLocationClient.requestLocationUpdates(locationRequest, callback, Looper.getMainLooper())
        } catch (e: SecurityException) {
            stopSelf()
        }
    }

    private fun stopTracking() {
        fusedLocationClient.removeLocationUpdates(callback)
        TrackingStore.stop()
        updateTrackingState()
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
    }

    private fun updateTrackingState() {
        val storeState = TrackingStore.state.value
        val coords = storeState.points.map { Pair(it.latitude, it.longitude) }
        _trackingState.value = TrackingState(
            isTracking = storeState.active,
            distanceMeters = storeState.distanceMeters,
            durationSeconds = storeState.durationSeconds,
            currentPaceSecondsPerKm = storeState.paceSecondsPerKm,
            routeCoordinates = coords
        )
    }

    private fun startForegroundServiceNotification() {
        val notification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("SPRINTA Tracking Active")
            .setContentText("Recording your GPS route and live metrics...")
            .setSmallIcon(android.R.drawable.ic_media_play)
            .setOngoing(true)
            .build()

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            startForeground(NOTIFICATION_ID, notification, android.content.pm.ServiceInfo.FOREGROUND_SERVICE_TYPE_LOCATION)
        } else {
            startForeground(NOTIFICATION_ID, notification)
        }
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(CHANNEL_ID, "SPRINTA Tracking Engine", NotificationManager.IMPORTANCE_LOW)
            getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
        }
    }

    override fun onDestroy() {
        fusedLocationClient.removeLocationUpdates(callback)
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null

    companion object {
        const val ACTION_START = "za.co.sprinta.app.action.START_TRACKING"
        const val ACTION_STOP = "za.co.sprinta.app.action.STOP_TRACKING"
        private const val CHANNEL_ID = "sprinta_tracking_engine"
        private const val NOTIFICATION_ID = 5102

        private val _trackingState = MutableStateFlow(TrackingState())
        val trackingState: StateFlow<TrackingState> = _trackingState
    }
}
package za.co.sprinta.app.tracking

import android.location.Location
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import za.co.sprinta.app.model.RoutePoint
import za.co.sprinta.app.util.GeoUtils

object TrackingStore {
    data class State(
        val active: Boolean = false,
        val startTimeMillis: Long = 0L,
        val distanceMeters: Double = 0.0,
        val durationSeconds: Long = 0L,
        val paceSecondsPerKm: Double = 0.0,
        val totalElevation: Double = 0.0,
        val points: List<RoutePoint> = emptyList()
    )

    private val _state = MutableStateFlow(State())
    val state: StateFlow<State> = _state

    fun start() {
        _state.value = State(
            active = true,
            startTimeMillis = System.currentTimeMillis()
        )
    }

    fun addLocation(location: Location) {
        val current = _state.value
        if (!current.active) return

        val newPoint = RoutePoint(
            latitude = location.latitude,
            longitude = location.longitude,
            altitude = location.altitude,
            timestamp = location.time
        )

        val previous = current.points.lastOrNull()
        val incrementalDistance = previous?.let {
            GeoUtils.haversineMeters(
                it.latitude,
                it.longitude,
                newPoint.latitude,
                newPoint.longitude
            )
        } ?: 0.0

        val elevationGain = previous?.let {
            maxOf(0.0, newPoint.altitude - it.altitude)
        } ?: 0.0

        val elapsed = ((System.currentTimeMillis() - current.startTimeMillis) / 1000L).coerceAtLeast(0L)
        val distance = current.distanceMeters + incrementalDistance
        val pace = GeoUtils.paceSecondsPerKm(distance, elapsed)

        _state.value = current.copy(
            distanceMeters = distance,
            durationSeconds = elapsed,
            paceSecondsPerKm = pace,
            totalElevation = current.totalElevation + elevationGain,
            points = current.points + newPoint
        )
    }

    fun stop() {
        val current = _state.value
        val elapsed = ((System.currentTimeMillis() - current.startTimeMillis) / 1000L).coerceAtLeast(current.durationSeconds)
        _state.value = current.copy(
            active = false,
            durationSeconds = elapsed,
            paceSecondsPerKm = GeoUtils.paceSecondsPerKm(current.distanceMeters, elapsed)
        )
    }

    fun reset() {
        _state.value = State()
    }
}

@OptIn(UnstableApi::class)
@Composable
fun VideoSplashScreen(
    onSplashComplete: () -> Unit
) {
    BackHandler(enabled = true) {}

    val context = LocalContext.current
    var isVideoLoaded by remember { mutableStateOf(false) }
    var videoFailed by remember { mutableStateOf(false) }

    val rawResId = remember { getRawVideoResId(context) }

    val exoPlayer = remember(context, rawResId) {
        if (rawResId != 0) {
            try {
                ExoPlayer.Builder(context).build().apply {
                    val videoUri = Uri.parse("android.resource://${context.packageName}/$rawResId")
                    setMediaItem(MediaItem.fromUri(videoUri))
                    volume = 0f
                    repeatMode = Player.REPEAT_MODE_OFF
                    prepare()
                    playWhenReady = true
                }
            } catch (e: Exception) {
                videoFailed = true
                null
            }
        } else {
            videoFailed = true
            null
        }
    }

    LaunchedEffect(Unit) {
        if (exoPlayer != null && !videoFailed) {
            exoPlayer.addListener(object : Player.Listener {
                override fun onPlaybackStateChanged(playbackState: Int) {
                    if (playbackState == Player.STATE_READY) {
                        isVideoLoaded = true
                    } else if (playbackState == Player.STATE_ENDED) {
                        onSplashComplete()
                    }
                }

                override fun onPlayerError(error: androidx.media3.common.PlaybackException) {
                    videoFailed = true
                }
            })
            delay(5000L)
            onSplashComplete()
        } else {
            delay(2500L)
            onSplashComplete()
        }
    }

    DisposableEffect(exoPlayer) {
        onDispose {
            exoPlayer?.release()
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(SprintaWhite),
        contentAlignment = Alignment.Center
    ) {
        if (exoPlayer != null && !videoFailed) {
            AndroidView(
                factory = { ctx ->
                    PlayerView(ctx).apply {
                        player = exoPlayer
                        useController = false
                        resizeMode = AspectRatioFrameLayout.RESIZE_MODE_ZOOM
                    }
                },
                modifier = Modifier.fillMaxSize()
            )
        }

        if (exoPlayer == null || videoFailed || !isVideoLoaded) {
            val alphaAnim = remember { Animatable(0f) }
            LaunchedEffect(Unit) {
                alphaAnim.animateTo(1f, animationSpec = tween(800))
            }

            Column(
                modifier = Modifier.alpha(alphaAnim.value),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                Image(
                    painter = painterResource(id = R.drawable.sprinta_flat_logo_transparent),
                    contentDescription = "SPRINTA Logo",
                    modifier = Modifier.size(160.dp)
                )
                Spacer(modifier = Modifier.height(16.dp))
                Text(
                    text = "SPRINTA",
                    style = MaterialTheme.typography.displaySmall.copy(fontWeight = FontWeight.Bold),
                    color = SprintaGold
                )
            }
        }
    }
}

private fun getRawVideoResId(context: Context): Int {
    val res1 = context.resources.getIdentifier("splash_video", "raw", context.packageName)
    if (res1 != 0) return res1
    val res2 = context.resources.getIdentifier("sprinta_splash", "raw", context.packageName)
    if (res2 != 0) return res2
    return 0
}

import android.location.Location
import java.util.Locale
import kotlin.math.atan2
import kotlin.math.cos
import kotlin.math.sin
import kotlin.math.sqrt

object LocationUtils {
    private const val EARTH_RADIUS_METERS = 6_371_000.0
    private const val MAX_ACCURACY_METERS = 15.0f
    private const val MAX_SPEED_MPS = 15.0 // 54 km/h

    fun haversineDistanceMeters(
        lat1: Double, lon1: Double,
        lat2: Double, lon2: Double
    ): Double {
        val dLat = Math.toRadians(lat2 - lat1)
        val dLon = Math.toRadians(lon2 - lon1)
        val a = sin(dLat / 2) * sin(dLat / 2) +
                cos(Math.toRadians(lat1)) * cos(Math.toRadians(lat2)) *
                sin(dLon / 2) * sin(dLon / 2)
        val c = 2 * atan2(sqrt(a), sqrt(1 - a))
        return EARTH_RADIUS_METERS * c
    }

    fun isReliableLocation(newLocation: Location, lastLocation: Location?): Boolean {
        if (newLocation.hasAccuracy() && newLocation.accuracy > MAX_ACCURACY_METERS) {
            return false
        }
        if (lastLocation != null) {
            val timeDiffSec = (newLocation.time - lastLocation.time) / 1000.0
            if (timeDiffSec > 0) {
                val distance = haversineDistanceMeters(
                    lastLocation.latitude, lastLocation.longitude,
                    newLocation.latitude, newLocation.longitude
                )
                val speed = distance / timeDiffSec
                if (speed > MAX_SPEED_MPS) {
                    return false
                }
            }
        }
        return true
    }

    fun formatPace(secondsPerKm: Double): String {
        if (secondsPerKm <= 0.0 || secondsPerKm.isInfinite() || secondsPerKm.isNaN()) {
            return "--:-- /km"
        }
        val totalSec = secondsPerKm.toInt()
        val minutes = totalSec / 60
        val seconds = totalSec % 60
        return "%02d:%02d /km".format(Locale.US, minutes, seconds)
    }

    fun formatDuration(seconds: Long): String {
        val h = seconds / 3600
        val m = (seconds % 3600) / 60
        val s = seconds % 60
        return if (h > 0) "%02d:%02d:%02d".format(Locale.US, h, m, s)
        else "%02d:%02d".format(Locale.US, m, s)
    }
}
package za.co.sprinta.app.util

import za.co.sprinta.app.data.local.PersonalRecordEntity
import za.co.sprinta.app.data.local.WorkoutEntity
import za.co.sprinta.app.model.NewRecord
import java.time.Instant

object PersonalRecordUtils {
    private data class Target(val type: String, val minimumDistanceMeters: Double)

    private val targets = listOf(
        Target("1K", 1_000.0),
        Target("5K", 5_000.0),
        Target("10K", 10_000.0)
    )

    fun detectNewRecords(
        workout: WorkoutEntity,
        existing: List<PersonalRecordEntity>
    ): List<NewRecord> {
        val newRecords = mutableListOf<NewRecord>()
        for (target in targets) {
            if (workout.distanceMeters < target.minimumDistanceMeters) continue
            val current = existing.firstOrNull { it.recordType == target.type }
            if (current == null || workout.averagePaceSecondsPerKm < current.valueSecondsPerKm) {
                newRecords += NewRecord(target.type, workout.averagePaceSecondsPerKm)
            }
        }
        return newRecords
    }

    fun toEntity(record: NewRecord, userId: String, workoutId: String): PersonalRecordEntity =
        PersonalRecordEntity(
            key = "$userId-${record.type}",
            userId = userId,
            recordType = record.type,
            valueSecondsPerKm = record.valueSecondsPerKm,
            workoutId = workoutId,
            achievedAtIso = Instant.now().toString()
        )
}

    private val passwordRegex = Regex("^(?=.*[A-Z])(?=.*\\d)(?=.*[^A-Za-z0-9]).{8,}$")

    fun isValidEmail(email: String): Boolean = emailRegex.matches(email.trim())
    fun isValidPassword(password: String): Boolean = passwordRegex.matches(password)
}

object GeoUtils {
    private const val EARTH_RADIUS_METERS = 6_371_000.0

    fun haversineMeters(
        lat1: Double,
        lon1: Double,
        lat2: Double,
        lon2: Double
    ): Double {
        val lat1Rad = Math.toRadians(lat1)
        val lat2Rad = Math.toRadians(lat2)
        val dLat = Math.toRadians(lat2 - lat1)
        val dLon = Math.toRadians(lon2 - lon1)
        val a = sin(dLat / 2) * sin(dLat / 2) +
            cos(lat1Rad) * cos(lat2Rad) * sin(dLon / 2) * sin(dLon / 2)
        val c = 2 * atan2(sqrt(a), sqrt(1 - a))
        return EARTH_RADIUS_METERS * c
    }

    fun paceSecondsPerKm(distanceMeters: Double, durationSeconds: Long): Double {
        if (distanceMeters <= 0.0) return 0.0
        return durationSeconds.toDouble() / (distanceMeters / 1000.0)
    }

    fun formatDuration(seconds: Long): String {
        val h = seconds / 3600
        val m = (seconds % 3600) / 60
        val s = seconds % 60
        return if (h > 0) "%02d:%02d:%02d".format(Locale.US, h, m, s)
        else "%02d:%02d".format(Locale.US, m, s)
    }

    fun formatPace(secondsPerKm: Double): String {
        if (secondsPerKm <= 0.0 || secondsPerKm.isInfinite() || secondsPerKm.isNaN()) return "--:-- /km"
        val total = secondsPerKm.toInt()
        return "%02d:%02d /km".format(Locale.US, total / 60, total % 60)
    }
}

import android.content.Context
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import androidx.work.Constraints
import androidx.work.NetworkType

import androidx.work.WorkManager
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import za.co.sprinta.app.AppGraph

class SyncWorker(
    appContext: Context,
    workerParams: WorkerParameters
) : CoroutineWorker(appContext, workerParams) {

    override suspend fun doWork(): Result = withContext(Dispatchers.IO) {
        val token = AppGraph.session.load()?.get("token")
            ?: return@withContext Result.failure()

        val unsynced = AppGraph.repository.getUnsyncedWorkouts()
        if (unsynced.isEmpty()) return@withContext Result.success()

        return@withContext try {
            for (workout in unsynced) {
                val response = AppGraph.api.syncWorkout(
                    "Bearer $token",
                    AppGraph.repository.entityToDto(workout)
                )

                when {
                    response.isSuccessful -> AppGraph.repository.markSynced(workout.workoutId)
                    response.code() in 500..599 -> return@withContext Result.retry()
                    response.code() == 401 -> return@withContext Result.failure()
                    else -> return@withContext Result.failure()
                }
            }
            Result.success()
        } catch (_: Exception) {
            Result.retry()
        }
    }

    companion object {
        fun schedule(context: Context) {
            val constraints = Constraints.Builder()
                .setRequiredNetworkType(NetworkType.CONNECTED)
                .build()

            val request = OneTimeWorkRequestBuilder<SyncWorker>()
                .setConstraints(constraints)
                .build()

            WorkManager.getInstance(context).enqueue(request)
        }
    }
}

@Composable
private fun TrackingScreen(
    vm: MainViewModel,
    navController: NavHostController,
    onFinishRun: () -> Unit,
    modifier: Modifier = Modifier
) {
    val serviceState by TrackingService.trackingState.collectAsState()
    val storeState by vm.trackingState.collectAsState()
    val prefs by vm.preferences.collectAsState()
    val context = LocalContext.current
    val coroutineScope = rememberCoroutineScope()

    var selectedActivityType by remember { mutableStateOf("RUN") }
    var mapType by remember { mutableStateOf(MapType.NORMAL) }

    var locationPermissionDenied by remember { mutableStateOf(false) }
    var showBackConfirmDialog by remember { mutableStateOf(false) }

    val permissionLauncher = rememberLauncherForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { permissions ->
        val granted = permissions[Manifest.permission.ACCESS_FINE_LOCATION] == true ||
                permissions[Manifest.permission.ACCESS_COARSE_LOCATION] == true
        if (granted) {
            startLocationService(context)
        } else {
            locationPermissionDenied = true
        }
    }

    val hasLocationPermission = ContextCompat.checkSelfPermission(
        context,
        Manifest.permission.ACCESS_FINE_LOCATION
    ) == PackageManager.PERMISSION_GRANTED || ContextCompat.checkSelfPermission(
        context,
        Manifest.permission.ACCESS_COARSE_LOCATION
    ) == PackageManager.PERMISSION_GRANTED

    val coords = if (serviceState.routeCoordinates.isNotEmpty()) {
        serviceState.routeCoordinates.map { LatLng(it.first, it.second) }
    } else {
        storeState.points.map { LatLng(it.latitude, it.longitude) }
    }

    val distanceMeters = if (serviceState.isTracking) serviceState.distanceMeters else storeState.distanceMeters
    val durationSeconds = if (serviceState.isTracking) serviceState.durationSeconds else storeState.durationSeconds
    val paceSecondsPerKm = if (serviceState.isTracking) serviceState.currentPaceSecondsPerKm else storeState.paceSecondsPerKm
    val isTrackingActive = serviceState.isTracking || storeState.active

    BackHandler(enabled = isTrackingActive) {
        showBackConfirmDialog = true
    }

    if (showBackConfirmDialog) {
        AlertDialog(
            onDismissRequest = { showBackConfirmDialog = false },
            title = {
                Text(
                    "Active Activity in Progress",
                    style = MaterialTheme.typography.titleLarge.copy(fontWeight = FontWeight.Bold),
                    color = MaterialTheme.colorScheme.onBackground
                )
            },
            text = {
                Text(
                    "Your activity is currently being recorded. Do you want to keep recording in the background or finish?",
                    style = MaterialTheme.typography.bodyLarge,
                    color = MaterialTheme.colorScheme.onBackground
                )
            },
            confirmButton = {
                Button(
                    onClick = {
                        showBackConfirmDialog = false
                        navController.popBackStack(NavScreen.Dashboard.route, inclusive = false)
                    },
                    colors = ButtonDefaults.buttonColors(
                        containerColor = SprintaGold,
                        contentColor = SprintaWhite
                    )
                ) {
                    Text("Keep Recording", fontWeight = FontWeight.Bold)
                }
            },
            dismissButton = {
                OutlinedButton(
                    onClick = {
                        showBackConfirmDialog = false
                        stopLocationService(context)
                        onFinishRun()
                    },
                    border = BorderStroke(1.dp, BorderGray)
                ) {
                    Text("Finish Activity", color = MaterialTheme.colorScheme.onBackground)
                }
            }
        )
    }

    Box(modifier = modifier.fillMaxSize()) {
        val initial = coords.lastOrNull() ?: LatLng(-33.9249, 18.4241)

        val cameraPositionState = rememberCameraPositionState {
            position = CameraPosition.fromLatLngZoom(initial, 17f)
        }

        LaunchedEffect(coords.lastOrNull()) {
            coords.lastOrNull()?.let { lastLocation ->
                cameraPositionState.animate(
                    update = CameraUpdateFactory.newLatLng(lastLocation),
                    durationMs = 1000
                )
            }
        }

        // Full Screen Map
        GoogleMap(
            modifier = Modifier.fillMaxSize(),
            cameraPositionState = cameraPositionState,
            properties = MapProperties(
                isBuildingEnabled = false,
                mapType = mapType,
                mapStyleOptions = if (mapType == MapType.NORMAL) MapStyleOptions.loadRawResourceStyle(context, R.raw.map_style_dark) else null
            )
        ) {
            if (coords.isNotEmpty()) {
                Polyline(
                    points = coords,
                    color = SprintaGold,
                    width = 14f,
                    jointType = JointType.ROUND,
                    startCap = RoundCap(),
                    endCap = RoundCap()
                )
                val lastPoint = coords.last()
                val markerState = rememberMarkerState(key = lastPoint.toString(), position = lastPoint)
                Marker(state = markerState, title = "Current Position")
            }
        }

        // Floating Glassmorphism HUD at Top
        Card(
            modifier = Modifier
                .align(Alignment.TopCenter)
                .padding(16.dp)
                .fillMaxWidth(),
            shape = RoundedCornerShape(20.dp),
            border = BorderStroke(1.dp, BorderGray),
            colors = CardDefaults.cardColors(containerColor = SprintaDarkSurface.copy(alpha = 0.9f))
        ) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(16.dp),
                horizontalArrangement = Arrangement.SpaceEvenly,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Text("TIME", style = MaterialTheme.typography.labelSmall, color = SprintaWhite.copy(alpha = 0.6f))
                    Text(
                        LocationUtils.formatDuration(durationSeconds),
                        style = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.Bold),
                        color = SprintaWhite
                    )
                }
                Box(
                    modifier = Modifier
                        .width(1.dp)
                        .height(30.dp)
                        .background(BorderGray)
                )
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Text("DISTANCE", style = MaterialTheme.typography.labelSmall, color = SprintaWhite.copy(alpha = 0.6f))
                    Text(
                        formatDistance(distanceMeters, prefs.useMiles),
                        style = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.Bold),
                        color = SprintaWhite
                    )
                }
                Box(
                    modifier = Modifier
                        .width(1.dp)
                        .height(30.dp)
                        .background(BorderGray)
                )
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Text("PACE", style = MaterialTheme.typography.labelSmall, color = SprintaWhite.copy(alpha = 0.6f))
                    Text(
                        LocationUtils.formatPace(paceSecondsPerKm),
                        style = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.Bold),
                        color = SprintaWhite
                    )
                }
            }
        }

        // Floating Action Buttons on Right
        Column(
            modifier = Modifier
                .align(Alignment.CenterEnd)
                .padding(end = 16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            FloatingActionButton(
                onClick = {
                    mapType = if (mapType == MapType.NORMAL) MapType.SATELLITE else MapType.NORMAL
                },
                containerColor = SprintaDarkSurface,
                contentColor = SprintaGold,
                shape = CircleShape,
                modifier = Modifier.size(48.dp)
            ) {
                Icon(Icons.Default.Layers, contentDescription = "Map Layers")
            }

            FloatingActionButton(
                onClick = {
                    coords.lastOrNull()?.let { lastLoc ->
                        coroutineScope.launch {
                            cameraPositionState.animate(CameraUpdateFactory.newLatLng(lastLoc))
                        }
                    }
                },
                containerColor = SprintaDarkSurface,
                contentColor = SprintaGold,
                shape = CircleShape,
                modifier = Modifier.size(48.dp)
            ) {
                Icon(Icons.Default.MyLocation, contentDescription = "Recenter")
            }
        }

        // Bottom Sheet Drawer & Controls
        Card(
            modifier = Modifier
                .align(Alignment.BottomCenter)
                .fillMaxWidth(),
            shape = RoundedCornerShape(topStart = 24.dp, topEnd = 24.dp),
            border = BorderStroke(1.dp, BorderGray),
            colors = CardDefaults.cardColors(containerColor = SprintaDarkSurface)
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(20.dp),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.spacedBy(16.dp)
            ) {
                if (locationPermissionDenied) {
                    Text(
                        stringResource(R.string.location_permission_required),
                        color = MaterialTheme.colorScheme.error
                    )
                }

                if (!isTrackingActive) {
                    // Activity Selection Chips
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.SpaceEvenly
                    ) {

private fun bestRecord(workouts: List<WorkoutEntity>, distanceMeters: Double): Double? {
    return workouts
        .filter { it.distanceMeters >= distanceMeters }
        .minByOrNull { it.averagePaceSecondsPerKm }
        ?.averagePaceSecondsPerKm
}

private fun shareBitmap(context: android.content.Context, bitmap: Bitmap) {
    try {
        val cachePath = File(context.cacheDir, "images")
        cachePath.mkdirs()
        val stream = FileOutputStream("$cachePath/sprinta_run.png")
        bitmap.compress(Bitmap.CompressFormat.PNG, 100, stream)
        stream.close()

        val imagePath = File(context.cacheDir, "images")
        val newFile = File(imagePath, "sprinta_run.png")
        val contentUri: Uri = FileProvider.getUriForFile(context, "${context.packageName}.provider", newFile)

        val shareIntent = Intent(Intent.ACTION_SEND).apply {
            type = "image/png"
            putExtra(Intent.EXTRA_STREAM, contentUri)
            addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
        }
        context.startActivity(Intent.createChooser(shareIntent, "Share your run"))
    } catch (e: Exception) {
        e.printStackTrace()
    }
}

private fun startLocationService(context: android.content.Context) {
    val intent = Intent(context, TrackingService::class.java).apply {
        action = TrackingService.ACTION_START
    }
    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) context.startForegroundService(intent)
    else context.startService(intent)
}

private fun stopLocationService(context: android.content.Context) {
    val intent = Intent(context, TrackingService::class.java).apply {
        action = TrackingService.ACTION_STOP
    }
    context.startService(intent)
}

private fun formatDate(isoString: String): String {
    return try {
        val sdf = SimpleDateFormat("yyyy-MM-dd'T'HH:mm:ss", Locale.US)
        val date = sdf.parse(isoString) ?: Date()
        SimpleDateFormat("EEE, d MMM yyyy", Locale.US).format(date)
    } catch (e: Exception) {
        isoString
    }
}

private fun formatDistance(meters: Double, miles: Boolean): String {
    val value = if (miles) meters / 1609.344 else meters / 1000.0
    return "%.2f %s".format(Locale.US, value, if (miles) "mi" else "km")
}

class MainViewModel(application: Application) : AndroidViewModel(application) {
    enum class Screen { AUTH, DASHBOARD, TRACKING, SUMMARY, SETTINGS, HISTORY, PRO_UPGRADE }

    private val repository = AppGraph.repository
    private val gson = Gson()

    private val _screen = MutableStateFlow(Screen.AUTH)
    val screen: StateFlow<Screen> = _screen

    private val _isRegistering = MutableStateFlow(false)
    val isRegistering: StateFlow<Boolean> = _isRegistering

    private val _busy = MutableStateFlow(false)
    val busy: StateFlow<Boolean> = _busy

    private val _message = MutableStateFlow<String?>(null)
    val message: StateFlow<String?> = _message

    private val _newRecords = MutableStateFlow<List<String>>(emptyList())
    val newRecords: StateFlow<List<String>> = _newRecords

    private val _session = MutableStateFlow(repository.currentSession())
    val session: StateFlow<Map<String, String>?> = _session

    val preferences: StateFlow<AppPreferences> = AppGraph.preferences.preferences
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), AppPreferences())

    val trackingState = TrackingStore.state

    val workouts: StateFlow<List<WorkoutEntity>> = _session
        .map { it?.get("id") }
        .flatMapLatest { userId ->
            if (userId == null) flowOf(emptyList()) else repository.observeWorkouts(userId)
        }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), emptyList())

    init {
        if (_session.value != null) {
            _screen.value = Screen.DASHBOARD
            SyncWorker.schedule(getApplication())
        }
    }

    fun setRegistering(value: Boolean) {
        _isRegistering.value = value
        _message.value = null
    }

    fun clearMessage() {
        _message.value = null
    }

    fun login(email: String, password: String) {
        if (!Validators.isValidEmail(email)) {
            _message.value = "Enter a valid email address."
            return
        }
        if (password.isBlank()) {
            _message.value = "Password is required."
            return
        }

        viewModelScope.launch {
            _busy.value = true
            val result = repository.login(email, password)
            _busy.value = false
            val response = result.getOrNull()
            if (response != null) {
                repository.saveSession(response)
                _session.value = repository.currentSession()
                _screen.value = Screen.DASHBOARD
                _message.value = null
                SyncWorker.schedule(getApplication())
                repository.refreshFromServer()
            } else {
                _message.value = if (result.exceptionOrNull()?.message?.contains("401") == true) {
                    "Login failed. Please check your details."
                } else {
                    "Unable to connect to SPRINTA server."
                }
            }
        }
    }

    fun register(name: String, email: String, password: String, confirmPassword: String) {
        when {
            name.isBlank() || email.isBlank() || password.isBlank() || confirmPassword.isBlank() -> {
                _message.value = "Please complete all required fields."
                return
            }
            !Validators.isValidEmail(email) -> {
                _message.value = "Enter a valid email address."
                return
            }
            !Validators.isValidPassword(password) -> {
                _message.value = "Password must be at least 8 characters and include an uppercase letter, a number and a special character."
                return
            }
            password != confirmPassword -> {
                _message.value = "Passwords do not match."
                return
            }
        }

        viewModelScope.launch {
            _busy.value = true
            val result = repository.register(name, email, password)
            _busy.value = false
            val response = result.getOrNull()
            if (response != null) {
                repository.saveSession(response)
                _session.value = repository.currentSession()
                _screen.value = Screen.DASHBOARD
                _isRegistering.value = false
                _message.value = null
                SyncWorker.schedule(getApplication())
                repository.refreshFromServer()
            } else {
                _message.value = "Registration failed. The email may already exist."
            }
        }
    }

    fun logout() {
        try {
            com.google.firebase.auth.FirebaseAuth.getInstance().signOut()
        } catch (_: Exception) {}
        repository.clearSession()
        _session.value = null
        _screen.value = Screen.AUTH
        TrackingStore.reset()
    }

    fun resetPassword(email: String) {
        if (!Validators.isValidEmail(email)) {
            _message.value = "Enter a valid email address."
            return
        }
        viewModelScope.launch {
            _busy.value = true
            try {
                com.google.firebase.auth.FirebaseAuth.getInstance().sendPasswordResetEmail(email.trim())
                _message.value = "Password reset link sent to $email."
            } catch (e: Exception) {
                _message.value = "Password reset link sent to $email."
            } finally {
                _busy.value = false
            }
        }
    }

    fun deleteAccount() {
        viewModelScope.launch {
            _busy.value = true
            try {
                com.google.firebase.auth.FirebaseAuth.getInstance().currentUser?.delete()
            } catch (_: Exception) {}
            repository.clearSession()
            _session.value = null
            _screen.value = Screen.AUTH
            TrackingStore.reset()
            _busy.value = false
            _message.value = "Your account has been deleted."
        }
    }

    fun goTo(screen: Screen) {
        _message.value = null
        _screen.value = screen
    }

    fun saveManualWorkout(
        activityType: String,
        distanceKm: Double,
        durationMinutes: Long,
        dateIso: String,
        notes: String
    ) {
        val userId = _session.value?.get("id") ?: return
        if (distanceKm <= 0.0 && durationMinutes <= 0L) {
            _message.value = "Please enter valid distance or duration."
            return
        }

        viewModelScope.launch {
            _busy.value = true
            val distanceMeters = distanceKm * 1000.0
            val durationSeconds = durationMinutes * 60L
            val pace = GeoUtils.paceSecondsPerKm(distanceMeters, durationSeconds)

            val entity = WorkoutEntity(
                workoutId = UUID.randomUUID().toString(),
                userId = userId,
                distanceMeters = distanceMeters,
                durationSeconds = durationSeconds,
                averagePaceSecondsPerKm = pace,
                startTimeIso = dateIso,
                totalElevation = 0.0,
                notes = notes.trim(),
                routePointsJson = "[]",
                activityType = activityType
            )
            val records = repository.saveWorkout(entity)
            _newRecords.value = records
            _busy.value = false
            SyncWorker.schedule(getApplication())
            _message.value = "Manual workout saved."
        }
    }

    fun saveCurrentWorkout(notes: String) {
        val userId = _session.value?.get("id") ?: return
        val state = TrackingStore.state.value
        if (state.points.size < 2 || state.distanceMeters <= 0.0) {
            _message.value = "Move a little further before saving your workout."
            return
        }

        viewModelScope.launch {
            _busy.value = true
            val entity = WorkoutEntity(
                workoutId = UUID.randomUUID().toString(),
                userId = userId,
                distanceMeters = state.distanceMeters,
                durationSeconds = state.durationSeconds,
                averagePaceSecondsPerKm = state.paceSecondsPerKm,
                startTimeIso = Instant.ofEpochMilli(state.startTimeMillis).toString(),
                totalElevation = state.totalElevation,
                notes = notes.trim(),
                routePointsJson = gson.toJson(state.points)
            )
            val records = repository.saveWorkout(entity)
            _newRecords.value = records
            _busy.value = false
            SyncWorker.schedule(getApplication())
            _message.value = "Workout saved."
            _screen.value = Screen.DASHBOARD
            TrackingStore.reset()
        }
    }

    fun resetRecordDialog() {
        _newRecords.value = emptyList()
    }

    fun setDarkMode(value: Boolean) {
        viewModelScope.launch { AppGraph.preferences.setDarkMode(value) }
    }

    fun setUseMiles(value: Boolean) {
        viewModelScope.launch { AppGraph.preferences.setUseMiles(value) }
    }

    fun setAudioInterval(value: Int) {
        viewModelScope.launch { AppGraph.preferences.setAudioInterval(value) }
    }

    fun setLanguage(value: String) {
        viewModelScope.launch { AppGraph.preferences.setLanguage(value) }
    }

    fun startSync() {
        SyncWorker.schedule(getApplication())
        _message.value = "Sync queued."
    }
}
